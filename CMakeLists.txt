cmake_minimum_required(VERSION 3.16)
project(CAN_Sentinel VERSION 1.0.0 LANGUAGES CXX)

option(CANSEC_BUILD_BENCHMARK "Build synthetic throughput benchmark" ON)
option(CANSEC_BUILD_FUZZERS "Build Clang libFuzzer targets" OFF)
option(CANSEC_BUILD_TESTS "Build C++ regression suite" ON)
option(CANSEC_ENABLE_SANITIZERS "Enable address and undefined behaviour sanitizers" OFF)
option(CANSEC_ENABLE_COVERAGE "Enable GCC/Clang gcov-compatible coverage" OFF)
option(CANSEC_WARNINGS_AS_ERRORS "Fail build on compiler warnings" ON)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

add_library(cansec_core STATIC
  src/frame.cpp src/parser.cpp src/signals.cpp src/rules.cpp
  src/engine.cpp src/report.cpp src/replay.cpp src/candump.cpp)
target_include_directories(cansec_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(MSVC)
  target_compile_options(cansec_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(cansec_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Wformat=2 -Wnull-dereference -Wnon-virtual-dtor)
  if(CANSEC_WARNINGS_AS_ERRORS)
    target_compile_options(cansec_core PRIVATE -Werror)
  endif()
endif()

if(NOT CANSEC_BUILD_FUZZERS)
  add_executable(cansec app/main.cpp)
  target_link_libraries(cansec PRIVATE cansec_core)
endif()
if(CANSEC_ENABLE_SANITIZERS AND NOT MSVC AND NOT CANSEC_BUILD_FUZZERS)
  foreach(target cansec_core cansec)
    target_compile_options(${target} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer -g)
    target_link_options(${target} PUBLIC -fsanitize=address,undefined)
  endforeach()
endif()
if(CANSEC_ENABLE_COVERAGE AND NOT MSVC AND NOT CANSEC_BUILD_FUZZERS)
  foreach(target cansec_core cansec)
    target_compile_options(${target} PUBLIC --coverage -O0 -g)
    target_link_options(${target} PUBLIC --coverage)
  endforeach()
endif()
if(CANSEC_BUILD_TESTS AND NOT CANSEC_BUILD_FUZZERS)
  enable_testing()
  add_executable(cansec_tests tests/test_main.cpp tests/test_frame.cpp tests/test_parser.cpp tests/test_signals.cpp tests/test_rules.cpp tests/test_engine.cpp tests/test_report.cpp tests/test_replay.cpp tests/test_candump.cpp)
  target_link_libraries(cansec_tests PRIVATE cansec_core)
  target_include_directories(cansec_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/tests)
  add_test(NAME cansec_unit COMMAND cansec_tests)
  if(CANSEC_ENABLE_SANITIZERS AND NOT MSVC AND NOT CANSEC_BUILD_FUZZERS)
    target_compile_options(cansec_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -g)
    target_link_options(cansec_tests PRIVATE -fsanitize=address,undefined)
  endif()
  if(CANSEC_ENABLE_COVERAGE AND NOT MSVC AND NOT CANSEC_BUILD_FUZZERS)
    target_compile_options(cansec_tests PRIVATE --coverage -O0 -g)
    target_link_options(cansec_tests PRIVATE --coverage)
  endif()
endif()
if(CANSEC_BUILD_BENCHMARK AND NOT CANSEC_BUILD_FUZZERS)
  add_executable(cansec_bench benchmarks/benchmark.cpp)
  target_link_libraries(cansec_bench PRIVATE cansec_core)
endif()
if(CANSEC_BUILD_FUZZERS)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "Fuzzer targets require Clang")
  endif()
  # The target library must be instrumented as well as the fuzzer entry points.
  target_compile_options(cansec_core PRIVATE -fsanitize=fuzzer-no-link,address,undefined -fno-omit-frame-pointer)
  add_executable(fuzz_candump defects/fuzz/fuzz_candump.cpp)
  target_link_libraries(fuzz_candump PRIVATE cansec_core)
  target_compile_options(fuzz_candump PRIVATE -fsanitize=fuzzer,address,undefined -fno-omit-frame-pointer)
  target_link_options(fuzz_candump PRIVATE -fsanitize=fuzzer,address,undefined)
  add_executable(fuzz_trace defects/fuzz/fuzz_trace.cpp)
  target_link_libraries(fuzz_trace PRIVATE cansec_core)
  target_compile_options(fuzz_trace PRIVATE -fsanitize=fuzzer,address,undefined -fno-omit-frame-pointer)
  target_link_options(fuzz_trace PRIVATE -fsanitize=fuzzer,address,undefined)
endif()
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND AND CANSEC_BUILD_TESTS AND NOT CANSEC_BUILD_FUZZERS)
  add_test(NAME cansec_cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:cansec>)
  add_test(NAME cansec_fault_campaign COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/evaluate_fault_campaign.py --binary $<TARGET_FILE:cansec>)
endif()
if(NOT CANSEC_BUILD_FUZZERS)
  install(TARGETS cansec RUNTIME DESTINATION bin)
endif()
